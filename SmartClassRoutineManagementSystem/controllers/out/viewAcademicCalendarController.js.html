  /**
   * @function getAllEvents
   * @description Fetch all academic events from the model, stripping out the 'id' field, and returning them in the response.
   * @param {Object} req - The request object.
   * @param {Object} res - The response object.
   * @returns {Promise<void>} Sends a JSON response with the events, or an error message in case of failure.
   */
  async getAllEvents(req, res) {
    try {
      // Fetch all events from the model
      const events = await AcademicCalendarModel.getAllEvents();

      // Strip out the 'id' field from the results
      const eventsWithoutId = events.map(event => {
        const { id, ...eventWithoutId } = event;
        return eventWithoutId; // Return the event object without the 'id' field
      });

      // Send the filtered events back in the response
      res.status(200).json(eventsWithoutId);
    } catch (error) {
      // Handle any errors that occur during the process
      console.error('Unexpected error in getAllEvents:', error);
      res.status(500).json({ error: 'Unexpected error occurred' });
    }
  },
