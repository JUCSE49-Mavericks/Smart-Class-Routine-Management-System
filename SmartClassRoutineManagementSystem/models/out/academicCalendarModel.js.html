const db = require('../config/db');
const moment = require('moment');

/**
 * Creates the academic_events table if it doesn't already exist.
 * @function
 */
const createAcademicEventsTable = () => {
    const query = `
        CREATE TABLE IF NOT EXISTS academic_events (
            id INT AUTO_INCREMENT PRIMARY KEY,
            event_name VARCHAR(255) NOT NULL,
            date DATE NOT NULL,
            department VARCHAR(255) NOT NULL,
            event_type ENUM('holiday', 'activity', 'vacation') NOT NULL
        );
    `;
    db.query(query, (err, results) => {
        if (err) {
            console.error('Error creating table:', err);
            throw err;
        }
        console.log('academic_events table created or already exists');
    });
};

/**
 * Retrieves all academic events from the database.
 * @function
 * @returns {Promise<Object[]>} A promise that resolves to an array of events.
 */
const getAllEvents = () => {
  return new Promise((resolve, reject) => {
     const query = `SELECT * FROM academic_events`;
     db.query(query, (err, results) => {
        if (err) {
           console.error('Error retrieving all events:', err);
           reject(err);
        } else {
           const formattedResults = results.map(event => {
             event.date = moment(event.date).format('YYYY-MM-DD'); // Format as 'YYYY-MM-DD'
             return event;
           });
           resolve(formattedResults);
        }
     });
  });
};

/**
 * Retrieves all holiday events from the academic_events table.
 * @function
 * @param {Function} callback - The callback function that handles the response.
 * @returns {void}
 */
const getHolidayEvents = (callback) => {
  const query = `SELECT * FROM academic_events WHERE event_type = 'holiday'`;
  db.query(query, (err, results) => {
    if (err) {
      console.error('Error retrieving holiday events:', err);
      callback(err, null);
      return;
    }

    // Format the results and remove the 'id' field
    const formattedResults = results.map(event => {
      const { id, ...eventWithoutId } = event; // Remove 'id' field
      eventWithoutId.date = moment(event.date).format('YYYY-MM-DD'); // Format the date
      return eventWithoutId;
    });

    callback(null, formattedResults);
  });
};

/**
 * Retrieves vacation events within a specified date range.
 * @function
 * @param {string} startDate - The start date of the range (YYYY-MM-DD).
 * @param {string} endDate - The end date of the range (YYYY-MM-DD).
 * @param {Function} callback - The callback function that handles the response.
 * @returns {void}
 */
const getVacations = (startDate, endDate, callback) => {
  const query = `
      SELECT * FROM academic_events 
      WHERE event_type = 'vacation' 
      AND date BETWEEN ? AND ?
  `;
  db.query(query, [startDate, endDate], (err, results) => {
      if (err) {
          console.error('Error retrieving vacations:', err);
          callback(err, null);
          return;

      // Remove 'id' field and format the date

  });
};
